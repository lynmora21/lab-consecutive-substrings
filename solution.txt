BEGIN

FUNCTION consecutiveSubstrings(string)

    CREATE empty list result

    FOR start = 0 TO length(string) - 1
        FOR end = start + 1 TO length(string)
            substring = substring(string, start, end)
            ADD substring TO result
        END FOR
    END FOR

    RETURN result

END FUNCTION

END
